## 4. Ejemplo: Test de Reducer

```javascript
import { describe, it, expect } from "vitest";
import pokemonReducer, { fetchPokemons } from "../pokemonSlice";

describe("pokemonSlice reducer", () => {
  it("should handle pending state", () => {
    const state = pokemonReducer(undefined, fetchPokemons.pending);
    expect(state.isLoading).toBe(true);
    expect(state.isError).toBe(false);
  });

  it("should handle fulfilled with data", () => {
    const action = {
      payload: { results: [{ id: 1, name: "bulbasaur" }], count: 1 },
    };
    const state = pokemonReducer(undefined, action);
    expect(state.pokemons).toHaveLength(1);
    expect(state.totalCount).toBe(1);
  });
});
```
